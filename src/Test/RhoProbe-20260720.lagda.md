Probe: the displaced ρ-line cost asymmetry. `triangleᴰ.ρ̂r` (234 ms)
vs `ρ̂l` (23 ms) — mirror terms, 10× apart. Two axes could differ:
which absorber slide the fill rides (`▾-idnᴰ` vs `emb-idn-absorbᴰ`)
and which side of `●ᴰ` the fill occupies. The controls form the 2×2:
`ctrl-swap-▾` puts a `▾-idnᴰ` fill on the right of `●ᴰ`,
`ctrl-swap-absorb` puts an `emb-idn-absorbᴰ` fill on the left. The
`-alone` rows measure the naked fills; `rho-r-named` re-runs the
whisker with the fill bound to a name first.

```agda
{-# OPTIONS --safe --erased-cubical --no-guardedness #-}

module Test.RhoProbe-20260720 where

open import Core.Type
open import Core.Base
open import Core.Data.Sigma
open import Core.Kan

open import Cat.Type
open import Cat.Base
open import Cat.Coherence
open import Cat.Displayed
open import Cat.Displayed.Base

module _ {o h o' h'} {C : category o h} (D : categoryᴰ C o' h') where
  open category C
  open theory C
  open categoryᴰ D
  open theoryᴰ D

  -- the whisker-of-fill combinator, elaborated in a fully general
  -- context: every piece is a variable, so the ▿ᴰ-head application
  -- stays neutral during its own elaboration
  ↝ᴰ-fill-●r
    : ∀ {x y z} {A B : composite x y} {F : composite y z} {x' y' z'}
        {U : is-representable A} {V : is-representable F}
        {e : A ≡ B}
        {A' : composite[ A ] x' y'} {B' : composite[ B ] x' y'}
        {F' : composite[ F ] y' z'}
      (U' : is-representable[ U ] A')
      (ê : PathP (λ i → composite[ e i ] x' y') A' B')
      (V' : is-representable[ V ] F')
      (m : I)
    → is-representable[ ↝-fill U e m ● V ] (ê m ▿ᴰ F')
  ↝ᴰ-fill-●r U' ê V' m = ↝ᴰ-fill U' ê m ●ᴰ V'

  module _ {x y z} {f : hom x y} {g : hom y z} {x' y' z'}
    (φ : hom[ f ] x' y') (ψ : hom[ g ] y' z')
    where

    private module T = triangle C f g

    ι₀ : hom[ idn x ] x' x'
    ι₀ = idn[ x' ]

    ι' : hom[ idn y ] y' y'
    ι' = idn[ y' ]

    ι'' : hom[ idn z ] z' z'
    ι'' = idn[ z' ]

    N' : composite[ T.A ▿ T.E ▿ T.B ] x' z'
    N' = emb[ φ ] ▿ᴰ emb[ ι' ] ▿ᴰ emb[ ψ ]

    ê₁ : PathP (λ i → composite[ T.e₁ i ] x' z') N' (emb[ φ ] ▿ᴰ emb[ ψ ])
    ê₁ i = emb[ φ ] ▿ᴰ emb-idn-absorbᴰ ψ i

    ê₂ : PathP (λ i → composite[ T.e₂ i ] x' z') N' (emb[ φ ] ▿ᴰ emb[ ψ ])
    ê₂ i = ▾-idnᴰ (emb[ φ ]) i ▿ᴰ emb[ ψ ]

    -- the verbatim pair
    rho-r : (m : I) → is-representable[ T.ρr m ] (ê₂ m)
    rho-r m = ↝ᴰ-fill (nrm[ φ ] ●ᴰ nrm[ ι' ]) (▾-idnᴰ (emb[ φ ])) m ●ᴰ nrm[ ψ ]

    rho-l : (m : I) → is-representable[ T.ρl m ] (ê₁ m)
    rho-l m = nrm[ φ ] ●ᴰ ↝ᴰ-fill (nrm[ ι' ] ●ᴰ nrm[ ψ ]) (emb-idn-absorbᴰ ψ) m

    -- the naked fills
    fill-▾-alone
      : (m : I)
      → is-representable[ ↝-fill (nrm f ● nrm (idn y)) (▾-idn (emb f)) m ]
          (▾-idnᴰ (emb[ φ ]) m)
    fill-▾-alone m = ↝ᴰ-fill (nrm[ φ ] ●ᴰ nrm[ ι' ]) (▾-idnᴰ (emb[ φ ])) m

    fill-absorb-alone
      : (m : I)
      → is-representable[ ↝-fill (nrm (idn y) ● nrm g) (emb-idn-absorb g) m ]
          (emb-idn-absorbᴰ ψ m)
    fill-absorb-alone m = ↝ᴰ-fill (nrm[ ι' ] ●ᴰ nrm[ ψ ]) (emb-idn-absorbᴰ ψ) m

    -- 2×2 controls: absorber × side
    ctrl-swap-▾
      : (m : I)
      → is-representable[ nrm f ● ↝-fill (nrm g ● nrm (idn z)) (▾-idn (emb g)) m ]
          (emb[ φ ] ▿ᴰ ▾-idnᴰ (emb[ ψ ]) m)
    ctrl-swap-▾ m = nrm[ φ ] ●ᴰ ↝ᴰ-fill (nrm[ ψ ] ●ᴰ nrm[ ι'' ]) (▾-idnᴰ (emb[ ψ ])) m

    ctrl-swap-absorb
      : (m : I)
      → is-representable[ ↝-fill (nrm (idn x) ● nrm f) (emb-idn-absorb f) m ● nrm g ]
          (emb-idn-absorbᴰ φ m ▿ᴰ emb[ ψ ])
    ctrl-swap-absorb m = ↝ᴰ-fill (nrm[ ι₀ ] ●ᴰ nrm[ φ ]) (emb-idn-absorbᴰ φ) m ●ᴰ nrm[ ψ ]

    -- the r-whisker with a named fill
    rho-r-named : (m : I) → is-representable[ T.ρr m ] (ê₂ m)
    rho-r-named m = fill-▾-alone m ●ᴰ nrm[ ψ ]

    -- the combinator instances
    rho-r-comb : (m : I) → is-representable[ T.ρr m ] (ê₂ m)
    rho-r-comb m = ↝ᴰ-fill-●r (nrm[ φ ] ●ᴰ nrm[ ι' ]) (▾-idnᴰ (emb[ φ ])) nrm[ ψ ] m

    ctrl-swap-absorb-comb
      : (m : I)
      → is-representable[ ↝-fill (nrm (idn x) ● nrm f) (emb-idn-absorb f) m ● nrm g ]
          (emb-idn-absorbᴰ φ m ▿ᴰ emb[ ψ ])
    ctrl-swap-absorb-comb m =
      ↝ᴰ-fill-●r (nrm[ ι₀ ] ●ᴰ nrm[ φ ]) (emb-idn-absorbᴰ φ) nrm[ ψ ] m
```
