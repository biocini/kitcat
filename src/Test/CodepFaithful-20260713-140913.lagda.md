## Layer A — fam-structure (memo §2)

The Petrakis substrate (following Petrakis, arXiv:2303.14754,
Def 2.1; entry resources/petrakis-dep-arrows/): hom/idn + ambient
base composition
(instantiated at the extracted center by the filling — no raw data
survives at the instance), abstract unary families with the two
actions, and the four 𝔣/cofam laws. Levels renamed from the memo
(`f r` → `fℓ rℓ`): the memo's level `f` collides with the
morphism binders in cofam₂/res-inv-l/codep₂-l.

```agda
record fam-structure {o h fℓ} (ob : Type o)
  : Type (o ⊔ h ₊ ⊔ fℓ ₊) where
  no-eta-equality
  field
    hom  : ob → ob → Type h
    idn  : (x : ob) → hom x x
    _⨾ᵇ_ : ∀ {x y z} → hom x y → hom y z → hom x z
    fam   : ob → Type fℓ
    cofam : ob → Type fℓ
    _◃_ : ∀ {y z} → hom y z → fam z → fam y
    _▹_ : ∀ {x y} → cofam x → hom x y → cofam y
    fam₁   : ∀ {y} (φ : fam y) → idn y ◃ φ ≡ φ
    fam₂   : ∀ {x y z} (g : hom x y) (h : hom y z) (φ : fam z)
           → (g ⨾ᵇ h) ◃ φ ≡ g ◃ (h ◃ φ)
    cofam₁ : ∀ {x} (c : cofam x) → c ▹ idn x ≡ c
    cofam₂ : ∀ {x y z} (c : cofam x) (f : hom x y) (g : hom y z)
           → c ▹ (f ⨾ᵇ g) ≡ (c ▹ f) ▹ g

  ctx : ob → ob → Type fℓ
  ctx x y = cofam x × fam y

  sub : ∀ {x y z} → hom y z → ctx x z → ctx x y
  sub g (c , φ) = c , g ◃ φ

  subl : ∀ {x y v} → hom x y → ctx x v → ctx y v
  subl f (c , φ) = c ▹ f , φ
```

## Layer B — codep-structure (memo §2)

`res` per context; the Π-integral composite; the honest
base-changing actions (defined, not fields — precomposition IS the
Def 4.1 application at the Π-integral — following Petrakis,
arXiv:2303.14754, Def 4.1, same entry); function-valued
res-invariance (identity at the filling — the memo's note 5

fam-law tracks; the fixed-endpoint actions.

```agda
record codep-structure {o h fℓ rℓ} {ob : Type o}
  (FS : fam-structure {o} {h} {fℓ} ob)
  : Type (o ⊔ h ⊔ fℓ ⊔ rℓ ₊) where
  no-eta-equality
  open fam-structure FS
  field
    res : ∀ {x y} → ctx x y → Type rℓ

  composite : ob → ob → Type (fℓ ⊔ rℓ)
  composite x y = (γ : ctx x y) → res γ

  _·ʰ_ : ∀ {x y z} → composite x y → (g : hom y z)
       → ((γ : ctx x z) → res (sub g γ))
  (F ·ʰ g) γ = F (sub g γ)

  _⟩ʰ_ : ∀ {x y v} (f : hom x y) → composite y v
       → ((γ : ctx x v) → res (subl f γ))
  (f ⟩ʰ F) γ = F (subl f γ)

  field
    res-inv-r : ∀ {x y z} (g : hom y z) (c : cofam x) (φ : fam z)
              → res (c , g ◃ φ) → res (c , φ)
    res-inv-l : ∀ {x y v} (f : hom x y) (c : cofam x) (φ : fam v)
              → res (c ▹ f , φ) → res (c , φ)
    codep₁-r : ∀ {x y} (c : cofam x) (φ : fam y)
             → PathP (λ i → res (c , fam₁ φ i) → res (c , φ))
                 (res-inv-r (idn y) c φ) (λ s → s)
    codep₂-r : ∀ {x y z w} (g : hom y z) (h : hom z w)
                 (c : cofam x) (φ : fam w)
             → PathP (λ i → res (c , fam₂ g h φ i) → res (c , φ))
                 (res-inv-r (g ⨾ᵇ h) c φ)
                 (λ s → res-inv-r h c φ (res-inv-r g c (h ◃ φ) s))
    codep₁-l : ∀ {x v} (c : cofam x) (φ : fam v)
             → PathP (λ i → res (cofam₁ c i , φ) → res (c , φ))
                 (res-inv-l (idn x) c φ) (λ s → s)
    codep₂-l : ∀ {x y z v} (f : hom x y) (g : hom y z)
                 (c : cofam x) (φ : fam v)
             → PathP (λ i → res (cofam₂ c f g i , φ) → res (c , φ))
                 (res-inv-l (f ⨾ᵇ g) c φ)
                 (λ s → res-inv-l f c φ (res-inv-l g (c ▹ f) φ s))

  _·_ : ∀ {x y z} → composite x y → hom y z → composite x z
  (F · g) (c , φ) = res-inv-r g c φ (F (c , g ◃ φ))
  infixl 30 _·_

  _⟩_ : ∀ {x y v} → hom x y → composite y v → composite x v
  (f ⟩ F) (c , φ) = res-inv-l f c φ (F (c ▹ f , φ))
```

```agda
  (CS : codep-structure {o} {h} {fℓ} {rℓ} FS)

  open codep-structure CS

    emb : ∀ {x y} → hom x y → composite x y
    compose-contr
      : ∀ {x y z} (f : hom x y) (g : hom y z)
      → is-contr (fiber emb (emb f · g))
    extract-agree
      : ∀ {x y z} (f : hom x y) (g : hom y z)
      → compose-contr f g .center .fst ≡ f ⨾ᵇ g

  f ⨾ g = compose-contr f g .center .fst
  infixr 40 _⨾_
```

## A1 — the tautological filling (memo §3)

Over `C : hcategory o h`, levels `fℓ := o ⊔ h`, `rℓ := h`. Record

```agda
module taut {o h} (C : hcategory o h) where
  open hcategory C

  FS : fam-structure {o} {h} {o ⊔ h} ob
  FS .fam-structure.hom  = hom
  FS .fam-structure.idn  = idn
  FS .fam-structure._⨾ᵇ_ = _⨾_
  FS .fam-structure.fam y   = Σ v ∶ ob , hom y v
  FS .fam-structure.cofam x = Σ w ∶ ob , hom w x
  FS .fam-structure._◃_ g φ = φ .fst , pre g (φ .snd)
  FS .fam-structure._▹_ c f = c .fst , post f (c .snd)
  FS .fam-structure.fam₁ φ     = ap (φ .fst ,_) (absorb-l (φ .snd))
  FS .fam-structure.fam₂ g h φ =
    ap (φ .fst ,_) (pre-comp g h (φ .snd))
  FS .fam-structure.cofam₁ c     = ap (c .fst ,_) (absorb-r (c .snd))
  FS .fam-structure.cofam₂ c f g =
    ap (c .fst ,_) (post-comp f g (c .snd))

  CS : codep-structure {o} {h} {o ⊔ h} {h} FS
  CS .codep-structure.res γ = hom (γ .fst .fst) (γ .snd .fst)
  CS .codep-structure.res-inv-r g c φ = λ s → s   -- ★ A1 (right)
  CS .codep-structure.res-inv-l f c φ = λ s → s   -- ★ A1 (left)
  CS .codep-structure.codep₁-r c φ     = refl     -- ★ A1c
  CS .codep-structure.codep₂-r g h c φ = refl     -- ★ A1c
  CS .codep-structure.codep₁-l c φ     = refl     -- ★ A1c
  CS .codep-structure.codep₂-l f g c φ = refl     -- ★ A1c

  RS : codep-representable CS
  RS .codep-representable.emb           = emb     -- ★ A1b
  RS .codep-representable.compose-contr = compose-contr
  RS .codep-representable.extract-agree f g = refl -- note 7

  module FS = fam-structure FS
  module CS = codep-structure CS
  module RS = codep-representable RS
```

```agda
    ap fst (is-contr→is-prop (ccL f g) lhs-pt (ccL f g .center))
```

## A3 — the engine over abstract non-refl res-inv (memo §4.A3)

res-inv-r is an opaque record field here — nothing reduces through
it. The left action never appears (the right-only firewall, memo A
finding 5). Baseline for the face cost count: Coherence's face₂₃
closes with exactly 2 `Path.assoc` bridges inside `reindex-face`.

Step 1: `·-comp` at the extraction — the only place res-inv cost
enters. Two links where Base pays one (`ap F ∘ happly sub-comp`),
but Base's link rests on sub-comp + pre-comp; the totals are
comparable, and at face level `·-comp` is an opaque path atom
exactly as in Base.

```agda
module A3 {o h fℓ rℓ} {ob : Type o}
  {FS : fam-structure {o} {h} {fℓ} ob}
  {CS : codep-structure {o} {h} {fℓ} {rℓ} FS}
  (RS : codep-representable CS)
  where
  open fam-structure FS
  open codep-structure CS
  open codep-representable RS

  ·-comp
    : ∀ {x y z w} (F : composite x y) (g : hom y z) (h : hom z w)
    → F · (g ⨾ h) ≡ (F · g) · h
  ·-comp {x} {w = w} F g h =
    funext λ γ → link₁ (γ .fst) (γ .snd) ∙ link₂ (γ .fst) (γ .snd)
    where
      ea : g ⨾ h ≡ g ⨾ᵇ h
      ea = extract-agree g h

      -- link₁: the extraction gap moved through both the coercion
      -- and the argument in one dependent line along ea (each i,
      -- ea i : hom y w types both); result type constant.
      link₁
        : (c : cofam x) (φ : fam w)
        → res-inv-r (g ⨾ h) c φ (F (c , (g ⨾ h) ◃ φ))
        ≡ res-inv-r (g ⨾ᵇ h) c φ (F (c , (g ⨾ᵇ h) ◃ φ))
      link₁ c φ i = res-inv-r (ea i) c φ (F (c , ea i ◃ φ))

      -- link₂: the codep₂-r PathP applied to the ap-F PathP over
      -- the SAME fam₂ track; result type constant, so the line is
      -- an ordinary path. The RHS is definitionally
      -- ((F · g) · h) (c , φ).
      link₂
        : (c : cofam x) (φ : fam w)
        → res-inv-r (g ⨾ᵇ h) c φ (F (c , (g ⨾ᵇ h) ◃ φ))
        ≡ res-inv-r h c φ
            (res-inv-r g c (h ◃ φ) (F (c , g ◃ (h ◃ φ))))
      link₂ c φ i =
```

```agda
    is-contr→is-prop (E₃-contr f g h) (pt-l f g h) (pt-r f g h)

        → (f ⨾ g) ⨾ h ≡ f ⨾ (g ⨾ h)
  assoc f g h = ap fst (assoc-σ f g h)
```
