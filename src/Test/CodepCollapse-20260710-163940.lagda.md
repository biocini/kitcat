Spike: two collapsed variants of the Cat.Codep coherence tower.

`collapse-AB` applies both moves: (A) uniform fiber prop-ness from the
record's own `is-representable-prop` (no per-target `subst` chains),
(B) single-whisker canonical witnesses routed through fiber lifts so
every face is `contr-face` against `ap (lift) assoc-σ`. `collapse-B`
applies Move B only: it keeps the `E₃-contr`/`E₄-contr` `subst` chains
verbatim and stays unit-free.

Measured against `Cat.Codep.Coherence`. Scratch only.

## Module 1 — collapse-AB (Move A + Move B)

```agda
module collapse-AB {o h} (C : codep-category o h) where
  open codep-category C

  -- Move A: fibers are props at every target, from the record itself.
  fib-prop : ∀ {x y} (F : composite x y) → is-prop (fiber emb F)
  fib-prop = is-representable-prop

  E₃ : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w) → composite x w
  E₃ f g h = emb f · g · h

  pt-l : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
       → fiber emb (E₃ f g h)
  pt-l f g h = (f ⨾ g) ⨾ h , emb-comp (f ⨾ g) h ∙ ap (_· h) (emb-comp f g)

  pt-r : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
       → fiber emb (E₃ f g h)
  pt-r f g h = f ⨾ (g ⨾ h) , emb-comp f (g ⨾ h) ∙ ·-comp (emb f) g h

  assoc-σ : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
          → pt-l f g h ≡ pt-r f g h
  assoc-σ f g h = fib-prop (E₃ f g h) (pt-l f g h) (pt-r f g h)

  assoc : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
        → (f ⨾ g) ⨾ h ≡ f ⨾ (g ⨾ h)
  assoc f g h = ap fst (assoc-σ f g h)

  module _ {x y z w v}
    (f : hom x y) (g : hom y z) (h : hom z w) (k : hom w v) where

    E₄ : composite x v
    E₄ = emb f · g · h · k

    -- The five pentagon vertices, single-whisker canonical witnesses.
    pt₁ : fiber emb E₄
    pt₁ = ((f ⨾ g) ⨾ h) ⨾ k
        , emb-comp ((f ⨾ g) ⨾ h) k
        ∙ ap (_· k) (emb-comp (f ⨾ g) h)
        ∙ ap (_· k) (ap (_· h) (emb-comp f g))

    pt₂ : fiber emb E₄
    pt₂ = (f ⨾ (g ⨾ h)) ⨾ k
        , emb-comp (f ⨾ (g ⨾ h)) k
        ∙ ap (_· k) (emb-comp f (g ⨾ h))
        ∙ ap (_· k) (·-comp (emb f) g h)

    pt₃ : fiber emb E₄
    pt₃ = f ⨾ ((g ⨾ h) ⨾ k)
        , emb-comp f ((g ⨾ h) ⨾ k)
        ∙ ·-comp (emb f) (g ⨾ h) k
        ∙ ap (_· k) (·-comp (emb f) g h)

    pt₄ : fiber emb E₄
    pt₄ = (f ⨾ g) ⨾ (h ⨾ k)
        , emb-comp (f ⨾ g) (h ⨾ k)
        ∙ ap (_· (h ⨾ k)) (emb-comp f g)
        ∙ ·-comp (emb f · g) h k

    pt₅ : fiber emb E₄
    pt₅ = f ⨾ (g ⨾ (h ⨾ k))
        , emb-comp f (g ⨾ (h ⨾ k))
        ∙ ·-comp (emb f) g (h ⨾ k)
        ∙ ·-comp (emb f · g) h k

    -- Move A: contractibility on demand, no per-target subst chain.
    E₄c : is-contr (fiber emb E₄)
    E₄c = prop-inhabited→is-contr (fib-prop E₄) pt₁

    σ₁₂ = fib-prop E₄ pt₁ pt₂
    σ₂₃ = fib-prop E₄ pt₂ pt₃
    σ₁₄ = fib-prop E₄ pt₁ pt₄
    σ₄₅ = fib-prop E₄ pt₄ pt₅
    σ₃₅ = fib-prop E₄ pt₃ pt₅

    α₁₂ = ap fst σ₁₂
    α₂₃ = ap fst σ₂₃
    α₁₄ = ap fst σ₁₄
    α₄₅ = ap fst σ₄₅
    α₃₅ = ap fst σ₃₅

    -- Right-whisker lift by k, left-whisker lift by f (via Φ), and the
    -- three reindex lifts. Every lift L satisfies fst ∘ L = reindex ∘ fst
    -- definitionally, so cong fst (ap L σ) is the intended associator.
    Λk : fiber emb (E₃ f g h) → fiber emb E₄
    Λk (m , p) = m ⨾ k , emb-comp m k ∙ ap (_· k) p

    Φ : composite y v → composite x v
    Φ L γ = emb f (γ .fst , (γ .snd .fst , L (ctr y , γ .snd)))

    Λf : fiber emb (E₃ g h k) → fiber emb E₄
    Λf (m , p) = f ⨾ m , emb-comp f m ∙ ap Φ p

    R₂₃ : fiber emb (E₃ f (g ⨾ h) k) → fiber emb E₄
    R₂₃ (m , p) = m , p ∙ ap (_· k) (·-comp (emb f) g h)

    R₄₅ : fiber emb (E₃ f g (h ⨾ k)) → fiber emb E₄
    R₄₅ (m , p) = m , p ∙ ·-comp (emb f · g) h k

    R₁₄ : fiber emb (E₃ (f ⨾ g) h k) → fiber emb E₄
    R₁₄ (m , p) = m , p ∙ ap (_· k) (ap (_· h) (emb-comp f g))

    -- face₁₂: right whisker. Bridges are single ap-comp distributions.
    face₁₂ : α₁₂ ≡ ap (_⨾ k) (assoc f g h)
    face₁₂ = contr-face E₄c σ₁₂ w₁₂ (λ i → Λk (assoc-σ f g h i)) v₁₂
      where
        w₁₂ : pt₁ .snd ≡ (Λk (pt-l f g h)) .snd
        w₁₂ = sym (ap (emb-comp ((f ⨾ g) ⨾ h) k ∙_)
          (ap-comp (_· k) (emb-comp (f ⨾ g) h) (ap (_· h) (emb-comp f g))))
        v₁₂ : (Λk (pt-r f g h)) .snd ≡ pt₂ .snd
        v₁₂ = ap (emb-comp (f ⨾ (g ⨾ h)) k ∙_)
          (ap-comp (_· k) (emb-comp f (g ⨾ h)) (·-comp (emb f) g h))

    -- face₃₅: left whisker via Φ. Bridges are single ap-comp Φ.
    face₃₅ : α₃₅ ≡ ap (f ⨾_) (assoc g h k)
    face₃₅ = contr-face E₄c σ₃₅ w₃₅ (λ i → Λf (assoc-σ g h k i)) v₃₅
      where
        w₃₅ : pt₃ .snd ≡ (Λf (pt-l g h k)) .snd
        w₃₅ = ap (emb-comp f ((g ⨾ h) ⨾ k) ∙_)
          (sym (ap-comp Φ (emb-comp (g ⨾ h) k) (ap (_· k) (emb-comp g h))))
        v₃₅ : (Λf (pt-r g h k)) .snd ≡ pt₅ .snd
        v₃₅ = ap (emb-comp f (g ⨾ (h ⨾ k)) ∙_)
          (ap-comp Φ (emb-comp g (h ⨾ k)) (·-comp (emb g) h k))

    -- face₂₃: reindex. Bridges are single Path.assoc reassociations.
    face₂₃ : α₂₃ ≡ assoc f (g ⨾ h) k
    face₂₃ = contr-face E₄c σ₂₃ w₂₃ (λ i → R₂₃ (assoc-σ f (g ⨾ h) k i)) v₂₃
      where
        w₂₃ : pt₂ .snd ≡ (R₂₃ (pt-l f (g ⨾ h) k)) .snd
        w₂₃ = Path.assoc (emb-comp (f ⨾ (g ⨾ h)) k)
          (ap (_· k) (emb-comp f (g ⨾ h))) (ap (_· k) (·-comp (emb f) g h))
        v₂₃ : (R₂₃ (pt-r f (g ⨾ h) k)) .snd ≡ pt₃ .snd
        v₂₃ = sym (Path.assoc (emb-comp f ((g ⨾ h) ⨾ k))
          (·-comp (emb f) (g ⨾ h) k) (ap (_· k) (·-comp (emb f) g h)))

    -- face₄₅: reindex. Bridges are single Path.assoc reassociations.
    face₄₅ : α₄₅ ≡ assoc f g (h ⨾ k)
    face₄₅ = contr-face E₄c σ₄₅ w₄₅ (λ i → R₄₅ (assoc-σ f g (h ⨾ k) i)) v₄₅
      where
        w₄₅ : pt₄ .snd ≡ (R₄₅ (pt-l f g (h ⨾ k))) .snd
        w₄₅ = Path.assoc (emb-comp (f ⨾ g) (h ⨾ k))
          (ap (_· (h ⨾ k)) (emb-comp f g)) (·-comp (emb f · g) h k)
        v₄₅ : (R₄₅ (pt-r f g (h ⨾ k))) .snd ≡ pt₅ .snd
        v₄₅ = sym (Path.assoc (emb-comp f (g ⨾ (h ⨾ k)))
          (·-comp (emb f) g (h ⨾ k)) (·-comp (emb f · g) h k))

    -- face₁₄: reindex; the risk point. w is reassoc, v needs the
    -- naturality square of ·-comp along emb-comp f g.
    face₁₄ : α₁₄ ≡ assoc (f ⨾ g) h k
    face₁₄ = contr-face E₄c σ₁₄ w₁₄ (λ i → R₁₄ (assoc-σ (f ⨾ g) h k i)) v₁₄
      where
        w₁₄ : pt₁ .snd ≡ (R₁₄ (pt-l (f ⨾ g) h k)) .snd
        w₁₄ = Path.assoc (emb-comp ((f ⨾ g) ⨾ h) k)
          (ap (_· k) (emb-comp (f ⨾ g) h))
          (ap (_· k) (ap (_· h) (emb-comp f g)))
        v₁₄ : (R₁₄ (pt-r (f ⨾ g) h k)) .snd ≡ pt₄ .snd
        v₁₄ = sym (Path.assoc A₁₄ N₁₄ C₁₄) ∙ ap (A₁₄ ∙_) nat₁₄
          where
            A₁₄ = emb-comp (f ⨾ g) (h ⨾ k)
            N₁₄ = ·-comp (emb (f ⨾ g)) h k
            C₁₄ = ap (_· k) (ap (_· h) (emb-comp f g))
            nat₁₄ = sym (homotopy-natural (λ F → ·-comp F h k) (emb-comp f g))

    -- The fiber-level pentagon and its projection.
    hom-identity : α₁₄ ∙ α₄₅ ≡ pcom (sym α₁₂) α₂₃ α₃₅
    hom-identity =
      coh-project E₄c fst (σ₁₄ ∙ σ₄₅) (pcom (sym σ₁₂) σ₂₃ σ₃₅)
        (ap-comp fst σ₁₄ σ₄₅)
        (pcom.ap (λ _ → fst) (sym σ₁₂) σ₂₃ σ₃₅)

    pentagon
      : assoc (f ⨾ g) h k ∙ assoc f g (h ⨾ k)
      ≡ ap (_⨾ k) (assoc f g h)
        ∙ assoc f (g ⨾ h) k
        ∙ ap (f ⨾_) (assoc g h k)
    pentagon =
      pcom (ap (_∙ α₄₅) face₁₄ ∙ ap (assoc (f ⨾ g) h k ∙_) face₄₅)
        hom-identity
        (λ i → pcom (sym (face₁₂ i)) (face₂₃ i) (face₃₅ i))
      ∙ pcom→∙
          (ap (_⨾ k) (assoc f g h))
          (assoc f (g ⨾ h) k)
          (ap (f ⨾_) (assoc g h k))
```

## Module 2 — collapse-B (Move B only, unit-free)

```agda
module collapse-B {o h} (C : codep-category o h) where
  open codep-category C

  pt-l : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
       → fiber emb (E₃ f g h)
  pt-l f g h = (f ⨾ g) ⨾ h , emb-comp (f ⨾ g) h ∙ ap (_· h) (emb-comp f g)

  pt-r : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
       → fiber emb (E₃ f g h)
  pt-r f g h = f ⨾ (g ⨾ h) , emb-comp f (g ⨾ h) ∙ ·-comp (emb f) g h

  -- Move B keeps the subst-manufactured per-target contractibility.
  E₃-contr : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
           → is-contr (fiber emb (E₃ f g h))
  E₃-contr f g h .center = pt-l f g h
  E₃-contr f g h .paths =
    is-contr→is-prop
      (subst (λ T → is-contr (fiber emb T))
        (ap (_· h) (emb-comp f g))
        (compose-contr (f ⨾ g) h)) _

  assoc-σ : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
          → pt-l f g h ≡ pt-r f g h
  assoc-σ f g h = is-contr→is-prop (E₃-contr f g h) (pt-l f g h) (pt-r f g h)

  assoc : ∀ {x y z w} (f : hom x y) (g : hom y z) (h : hom z w)
        → (f ⨾ g) ⨾ h ≡ f ⨾ (g ⨾ h)
  assoc f g h = ap fst (assoc-σ f g h)

        emb-comp ((f ⨾ g) ⨾ h) k
      ∙ ap (_· k) (emb-comp (f ⨾ g) h)
      ∙ ap (_· k) (ap (_· h) (emb-comp f g))

      is-contr→is-prop
        (subst (λ T → is-contr (fiber emb T)) path₄
          (compose-contr ((f ⨾ g) ⨾ h) k)) _
      where
        path₄ : emb ((f ⨾ g) ⨾ h) · k ≡ E₄
        path₄ = ap (_· k) (emb-comp (f ⨾ g) h)
              ∙ ap (_· k) (ap (_· h) (emb-comp f g))

    pt₁ : fiber emb E₄
    pt₁ = ((f ⨾ g) ⨾ h) ⨾ k
        , emb-comp ((f ⨾ g) ⨾ h) k
        ∙ ap (_· k) (emb-comp (f ⨾ g) h)
        ∙ ap (_· k) (ap (_· h) (emb-comp f g))

    pt₂ : fiber emb E₄
    pt₂ = (f ⨾ (g ⨾ h)) ⨾ k
        , emb-comp (f ⨾ (g ⨾ h)) k
        ∙ ap (_· k) (emb-comp f (g ⨾ h))
        ∙ ap (_· k) (·-comp (emb f) g h)

    pt₃ : fiber emb E₄
    pt₃ = f ⨾ ((g ⨾ h) ⨾ k)
        , emb-comp f ((g ⨾ h) ⨾ k)
        ∙ ·-comp (emb f) (g ⨾ h) k
        ∙ ap (_· k) (·-comp (emb f) g h)

    pt₄ : fiber emb E₄
    pt₄ = (f ⨾ g) ⨾ (h ⨾ k)
        , emb-comp (f ⨾ g) (h ⨾ k)
        ∙ ap (_· (h ⨾ k)) (emb-comp f g)

        , emb-comp f (g ⨾ (h ⨾ k))
        ∙ ·-comp (emb f) g (h ⨾ k)

    σ₄₅ = is-contr→is-prop E₄c pt₄ pt₅
```
